      data_donat = data_donat_param.map(m => ({ ...m, status: m.hasil_kesimpulan }))
    }
    if (data_donat_param && data_donat_param.length > 0 && kk == 'toe') {
      data_donat = data_donat_param.map(m => ({ ...m, status: m.hasil_uji_keseluruhan }))
    }
    if (data_donat && data_donat.length > 0) {

      const id_status_labels = Object.keys(efektif_tidakefektif_tat_idx).map(m => m)
      const datasetsitem = {
        label: '',
        backgroundColor: id_status_labels.map(m => efektif_tidakefektif_tat_hex[m]),
        hoverOffset: 4,
      }

      let datac = []
      for (let m of id_status_labels) {
        datac.push(0)
      }


      let is_status_null_already = false
      for (let m of data_donat) {
        if (m.hasOwnProperty('status_kosong')) {
          is_status_null_already = true
          break
        }
      }

      if (is_activated_status_null && is_status_null_already) {
        donat.labels.push('Dalam Proses')
        datasetsitem.backgroundColor.push(colorsPallette.lightgrey)
        datac.push(0)
      }

      for (let m of data_donat) {
        let idx_of_labels = id_status_labels.indexOf(`${m.status}`)
        datac[idx_of_labels] = m.jumlah
        donat.total += m.jumlah

        if (is_activated_status_null && is_status_null_already && !m.status_kosong) {
          datac[datac.length - 1] = m.jumlah
        }
      }

      datasetsitem.data = datac
      donat.datasets.push(datasetsitem)

    }

    return donat
  }

      for (let m of data_donat) {
        if (m.hasOwnProperty('status_kosong')) {
          is_status_null_already = true
          break
        }
      }

      if (is_activated_status_null && is_status_null_already) {
        donat.labels.push('Dalam Proses')
        datasetsitem.backgroundColor.push(colorsPallette.lightgrey)
        datac.push(0)
      }

      for (let m of data_donat) {
        let idx_of_labels = id_status_labels.indexOf(`${m.status}`)
        datac[idx_of_labels] = m.jumlah
        donat.total += m.jumlah

        if (is_activated_status_null && is_status_null_already && !m.status_kosong) {
          datac[datac.length - 1] = m.jumlah
        }
      }

      datasetsitem.data = datac
      donat.datasets.push(datasetsitem)

    }

    return donat
  }
