  getmaxperiode = async (filtersave) => {
    const { datafilter, other_state } = this.state
    const { getapi_services } = api_services({ api_path: `/get_max_periode` })

    this.setState({ is_loading: true })
    const response = await getapi_services({
      filter: {
        ...datafilter,
        filter: {
          id_ruang_lingkup: other_state.id_ruang_lingkup
        },
      },
    })

    this.setState({ is_loading: false })
    if (response.code || response.error) return
    if (response.data.length == 0) return
    let id_periode = response.data[response.data.length - 1].id_periode

    if (filtersave && filtersave['id_periode']) {
      id_periode = filtersave['id_periode']
    }

    if (other_state.id_periode) {
      id_periode = other_state.id_periode
    }

    this.setState(state => ({
      other_state: {
        ...state.other_state,
        id_periode
      }
    }), this.gets)
  }

  handlegetperiode = async (inputValue, callback) => {

        ...this.state.datafilter,
        filter: {
          id_ruang_lingkup: other_state.id_ruang_lingkup

      }
    }
    datafilter.filter.filter.not_draft = 1
    if (inputValue && callback) {
      datafilter.filter.nama = inputValue
    }
    const dataarr = await this.periodeModel.getCombo(datafilter)
    if (dataarr.error || dataarr.code) return

    if (inputValue && callback) {
      callback(dataarr)

        ...state.listcombo,
        id_periode: dataarr
      }
    }))

  }

  getdonat = (kk, data_donat) => {
    const { other_state } = this.state
    let donat = {}
    donat.labels = other_state[kk].donat.labels
    if (kk == 'too') {
      donat.labels = other_state['tod'].donat.labels
    }
    donat.datasets = []
    donat.total = 0

    // console.log('GETDONAT' + kk);
    // console.log(donat);



    if (data_donat && data_donat.length > 0) {

      const id_status_labels = []
      for (let m of other_state[kk].donat.labels) {
        // id_status_labels.push(Object.keys(status_pengajuan_hardcode).filter(x => status_pengajuan_hardcode[x] == m ? x : '')[0])
        id_status_labels.push(Object.keys(this.state.listreferensi.id_status).filter(x => this.state.listreferensi.id_status[x] == m ? x : '')[0])
      }
      let datastatus_line_arr = kk == 'csa' ? status_line_csa_arr : kk == 'tod' ? status_line_tod_arr : status_line_toe_arr
      if (kk == 'too') {
        datastatus_line_arr = status_line_tod_arr
      }

      const color_pengajuan = {
        // ...color_pengajuan_hardcode_hex
        ...this.state.listreferensi.id_status_color
      }

      if (is_activated_first_line_status_color_grey) {
        if (kk == 'tod' || kk == 'too') {
          color_pengajuan['6'] = colorsPallette.grey
        }
        if (kk == 'toe') {
          color_pengajuan['10'] = colorsPallette.grey

        }
      }

      const datasetsitem = {
        label: '',
        backgroundColor: datastatus_line_arr.map(m => color_pengajuan[m]),
        hoverOffset: 4,
      }
      let datac = []
      for (let m of id_status_labels) {
        datac.push(0)
      }
      for (let m of data_donat) {
        let idx_of_labels = id_status_labels.indexOf(`${m.id_status}`)
        datac[idx_of_labels] = m.jumlah
        donat.total += m.jumlah
      }
      datasetsitem.data = datac
      donat.datasets.push(datasetsitem)

    }

    return donat
  }

const PieOrBar = props => {

  // console.log('PieOrBar');
  // console.log(props);
  const chartRef = useRef();
  const { is_bar, data, title, ondetail } = props

  const centerTextPlugin = (total) => {
    return {
      id: 'centerText',
      afterDraw: (chart) => {
        const { width, height } = chart;
        const { top, bottom, left, right } = chart.chartArea;

        const ctx = chart.ctx;
        ctx.save();

        const fontSize = (height / 150).toFixed(2);
        ctx.font = `${fontSize}em sans-serif`;
        ctx.textBaseline = 'middle';
        ctx.textAlign = 'center';

        const text = total?.toString() || '0';

        const textX = (left + right) / 2;
        const textY = (top + bottom) / 2;

        ctx.fillText(text, textX, textY);
        ctx.restore();
      },
    }
  }



  if (is_bar && data && Object.keys(data).length > 0) {
    return (
      <Bar
        data={data}
        options={optiondonat}
        height={70}
      />
    )
  }
  if (!is_bar && data && Object.keys(data).length > 0 && data.datasets.length > 0) {
    return (
      <>
        <Doughnut
          ref={chartRef}
          options={{
            ...optiondonat,
            plugins: {
              ...optiondonat.plugins,
              title: {
                ...optiondonat.plugins.title,
                text: title
              }
            }
          }}
          data={data}
          plugins={data.datasets.length > 0 ? [centerTextPlugin(data.total)] : []}
        // height={300}
        // width={300}
        />
        {ondetail && data && data.datasets && data.datasets.length > 0 ? (
          <div className="d-flex justify-content-center mt-3 mb-3">
            <Button
              className="btn-default-app"
              disabled={false}
              onClick={ondetail}
            >
              Detail
              <span className="material-icons icon-btn-left ms-1">
                chevron_right
              </span>
            </Button>
          </div>
        ) : null}
      </>

    )
  }
  return null
}

  if (props.other_state.jenis_progress_efektivitas == 'efektivitas' && props.other_state.on_detail_line == 'remediasi') {
    heads = head_efektivitas.map(m => m == 'Efektivitas' ? 'Kesimpulan' : m)
    if (props.other_state.id_ruang_lingkup > 5) {
      heads = head_efektivitas.map(m => m == 'Efektivitas' ? 'Kesimpulan' : m).filter(m => m != 'No Sub Proses')
    }
  }

  return (
    <div className="row">
      <div className="col-md-12">
        <table className='w-full table table-auto border-collapse border'>
          <thead>
            <tr>
              <th className="border" style={{ width: 1 }}>No.</th>
              {heads.map((m, i) => (
                <th key={i} className="border">{m}</th>
              ))}
            </tr>
          </thead>
          {props.other_state.jenis_progress_efektivitas == 'progres' ? (

            <tbody>
              {props.other_state.on_detail_line == 'kuesioner' ? (
                <>
                  {props.other_state.unit_detail_detail.map((m, i) => (
                    <tr key={i}>
                      <td className="border text-center">{i + 1}</td>
                      <td className="border">{m.ref_kontrol}</td>
                      {props.other_state.id_ruang_lingkup > 5 ? null : (
                        <>
                          <td className="border">{m.no_sub_proses}</td>
                          <td className="border ">{m.ref_proses_lvl4}</td>
                          <td className="border ">{m.ref_proses_lvl5}</td>
                        </>
                      )}
                      <td className="border ">{m.control_preparer}</td>
                      <td className="border">{m.status == 'draft' ? 'Draft' : 'Sudah Kuesioner'}</td>
                    </tr>
                  ))}
                </>
              ) : props.other_state.on_detail_line == 'csa' ? (
                <>
                  {props.other_state.unit_detail_detail.map((m, i) => (
                    <tr key={i}>
                      <td className="border text-center">{i + 1}</td>
                      <td className="border">{m.ref_kontrol}</td>
                      {props.other_state.id_ruang_lingkup > 5 ? null : (
                        <>
                          <td className="border">{m.no_sub_proses}</td>
                          <td className="border ">{m.ref_proses_lvl4}</td>
                          <td className="border ">{m.ref_proses_lvl5}</td>
                        </>
                      )}
                      <td className="border">{m.control_preparer}</td>
                      {/* <td className="border">{status_pengajuan_hardcode[m.id_status]}</td> */}
                      <td className="border">{props.listreferensi.id_status[m.id_status]}</td>
                    </tr>
                  ))}
                </>
              ) : props.other_state.on_detail_line == 'tod' ? (
                <>
                  {props.other_state.unit_detail_detail.map((m, i) => (
                    <tr key={i}>
                      <td className="border text-center">{i + 1}</td>
                      <td className="border">
                        <div dangerouslySetInnerHTML={{ __html: stringEnter(m.ref_kontrol) }}></div>
                      </td>
                      {props.other_state.id_ruang_lingkup > 5 ? null : (
                        <>
                          <td className="border">
                            <div dangerouslySetInnerHTML={{ __html: stringEnter(m.no_sub_proses) }}></div>
                          </td>
                        </>
                      )}
                      <td className="border">
                        <div dangerouslySetInnerHTML={{ __html: stringEnter(m.control_preparer_jabatan) }}></div>
                      </td>
                      <td className="border">
                        <div dangerouslySetInnerHTML={{ __html: stringEnter(m.nama_subbid) }}></div>
                      </td>
                      {/* <td className="border">{status_pengajuan_hardcode[m.id_status]}</td> */}
                      <td className="border">{props.listreferensi.id_status[m.id_status]}</td>
                    </tr>
                  ))}
                </>
              ) : props.other_state.on_detail_line == 'toe' ? (
                <>
                  {props.other_state.unit_detail_detail.map((m, i) => (
                    <tr key={i}>
                      <td className="border text-center">{i + 1}</td>
                      <td className="border">{m.ref_kontrol}</td>
                      <td className="border">{m.frekuensi}</td>
                      {m.data_periode && Object.keys(m.data_periode).length > 0 ? (
                        <>
                          {Object.keys(m.data_periode).map((x, y) => (
                            <td className="border text-end" key={y}>{m.data_periode[x].jumlah_working_paper}</td>
                          ))}
                        </>
                      ) : null}
                      <td className="border text-end">{m.jumlah_sampel}</td>
                      <td className="border text-end">{m.standard_sampel}</td>
                      <td className="border text-end">{m.kekurangan_sampel}</td>
                      <td className="border">{props.listreferensi.id_status[m.id_status]}</td>
                    </tr>
                  ))}
                </>
              ) : null}
            </tbody>
          ) : (

            <tbody>
              {['csa', 'tod'].includes(props.other_state.on_detail_line) ? (
                <>
                  {props.other_state.unit_detail_detail.map((m, i) => (
                    <tr key={i}>
                      <td className="border text-center">{i + 1}</td>
                      <td className="border">{m.ref_kontrol}</td>
                      {props.other_state.id_ruang_lingkup > 5 ? null : (
                        <>
                          <td className="border">{m.no_sub_proses}</td>
                          <td className="border">{m.ref_proses_lvl4}</td>
                          <td className="border">{m.ref_proses_lvl5}</td>
                        </>
                      )}
                      <td className="border">{props.other_state.on_detail_line == 'tod' ? m.control_preparer_jabatan : m.control_preparer}</td>
                      {props.other_state.on_detail_line == 'tod' ? (
                        <td className="border">
                          <div dangerouslySetInnerHTML={{ __html: stringEnter(m.nama_subbid) }}></div>
                        </td>
                      ) : null}
                      <td className="border">{efektif_tidakefektif_tat_idx[props.other_state.on_detail_line == 'csa' ? m.efektivitas_csa : m.hasil_kesimpulan]}</td>

                    </tr>
                  ))}
                </>
              ) : props.other_state.on_detail_line == 'too' ? (
                <>
                  {props.other_state.unit_detail_detail.map((m, i) => (
                    <tr key={i}>
                      <td className="border text-center">{i + 1}</td>
                      {props.other_state.id_ruang_lingkup > 5 ? null : (
                        <>
                          <td className="border">{m.no_sub_proses}</td>
                        </>
                      )}
                      <td className="border">{m.ref_kontrol}</td>
                      <td className="border">{m.lokasi}</td>
                      <td className="border">{efektif_tidakefektif_tat_idx[props.other_state.on_detail_line == 'csa' ? m.efektivitas_csa : m.hasil_kesimpulan]}</td>

                    </tr>
                  ))}
                </>
              ) : props.other_state.on_detail_line == 'toe' ? (
                <>
                  {props.other_state.unit_detail_detail.map((m, i) => (
                    <tr key={i}>
                      <td className="border text-center">{i + 1}</td>
                      <td className="border">{m.ref_kontrol}</td>
                      <td className="border">{m.frekuensi}</td>
                      <td className="border">{efektif_tidakefektif_tat_idx[m.hasil_uji_keseluruhan]}</td>

                    </tr>
                  ))}
                </>
              ) : props.other_state.on_detail_line == 'dod' ? (
                <>
                  {props.other_state.unit_detail_detail.map((m, i) => (
                    <tr key={i}>
                      <td className="border text-center">{i + 1}</td>
                      {props.other_state.id_ruang_lingkup > 5 ? null : (
                        <>
                          <td className="border">{m.no_sub_proses}</td>
                        </>
                      )}
                      <td className="border">{m.ref_kontrol}</td>
                      <td className="border">{m.lokasi}</td>
                      <td className="border">{m.kesimpulan_defisiensi}</td>

                    </tr>
                  ))}
                </>
              ) : props.other_state.on_detail_line == 'remediasi' ? (
                <>
                  {props.other_state.unit_detail_detail.map((m, i) => (
                    <tr key={i}>
                      <td className="border text-center">{i + 1}</td>
                      {props.other_state.id_ruang_lingkup > 5 ? null : (
                        <>
                          <td className="border">{m.no_sub_proses}</td>
                        </>
                      )}
                      <td className="border">{m.ref_kontrol}</td>
                      <td className="border">{m.lokasi}</td>
                      <td className="border">{status_remediasi_edit_idx[m.status_remediasi]}</td>

                    </tr>
                  ))}
                </>
              ) : null}
            </tbody>

          )}
        </table>
      </div>
    </div>
  )
}
